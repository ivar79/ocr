    from inference import OCREngine
    
    ocr = OCREngine()
    
    # تصویر
    result = ocr.read_image('image.png')
    print("متن:", result['text'])
    
    # یا PDF
    result = ocr.read('document.pdf')
    for page in result['pages']:
        print(page['text'])

📋 مثال عملی 1: استخراج متن از تصویر فارسی
════════════════════════════════════════════════════════════════════

from inference import OCREngine

ocr = OCREngine()
result = ocr.read_image('farsi.png')

print("متن استخراج‌شده:")
print(result['text'])

# ذخیره در فایل
with open('output.txt', 'w', encoding='utf-8') as f:
    f.write(result['text'])

📋 مثال عملی 2: پردازش تمام تصاویر در پوشه
════════════════════════════════════════════════════════════════════

from inference import OCREngine
from pathlib import Path

ocr = OCREngine()

for image in Path('.').glob('*.png'):
    result = ocr.read_image(str(image))
    
    # ذخیره متن
    with open(image.stem + '.txt', 'w', encoding='utf-8') as f:
        f.write(result['text'])
    
    print(f"✓ {image.name}")

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📋 مثال عملی 3: استخراج متن از PDF
════════════════════════════════════════════════════════════════════

from inference import OCREngine

ocr = OCREngine()
result = ocr.read('document.pdf')

print(f"تعداد صفحات: {len(result['pages'])}")

for page in result['pages']:
    print(f"صفحه {page.get('page_num')}:")
    print(page['text'])
    print("---")

🌐 مثال عملی 4: استفاده از API با Curl
════════════════════════════════════════════════════════════════════

ابتدا سرور API را شروع کن:
python api/app.py

سپس در ترمینال دیگر:

# تصویر بفرست
curl -X POST "http://localhost:8000/ocr/image" ^
  -H "Content-Type: multipart/form-data" ^
  -F "file=@image.png"

# PDF بفرست
curl -X POST "http://localhost:8000/ocr/pdf" ^
  -H "Content-Type: multipart/form-data" ^
  -F "file=@document.pdf"

# بررسی وضعیت
curl http://localhost:8000/health
